# ModemCheck Cloud Server - Cron Configuration Example
#
# Add these entries to your crontab to automate maintenance tasks.
#
# Installation:
#   crontab -e
#   (paste the relevant entries below)
#
# Note: Update paths to match your installation directory

# ============================================================================
# AUTOMATED BACKUPS
# ============================================================================

# Daily full backup at 2:00 AM (PostgreSQL + Redis with verification)
0 2 * * * cd /path/to/modemcheck/cloudserver && ./backup-all.sh --verify >> logs/backup.log 2>&1

# Alternative: Separate backup schedules
# Daily PostgreSQL backup at 2:00 AM
# 0 2 * * * cd /path/to/modemcheck/cloudserver && ./backup-database.sh --verify >> logs/backup-postgres.log 2>&1

# Daily Redis backup at 2:30 AM
# 30 2 * * * cd /path/to/modemcheck/cloudserver && ./backup-redis.sh >> logs/backup-redis.log 2>&1


# ============================================================================
# CONFIG MANAGEMENT - NONCE CLEANUP (CRITICAL)
# ============================================================================

# Hourly nonce cleanup (prevents config_nonces table bloat)
# CRITICAL: Without this, the nonces table grows ~5000 rows/hour at scale
15 * * * * cd /path/to/modemcheck/cloudserver && source .venv/bin/activate && python3 scripts/cleanup_nonces.py >> logs/nonce-cleanup.log 2>&1


# ============================================================================
# CONFIG MANAGEMENT - PARTITION MAINTENANCE
# ============================================================================

# Monthly partition creation (run on 1st of each month at 2:00 AM)
# Creates config_audit_logs partition for the next month
0 2 1 * * /path/to/modemcheck/cloudserver/scripts/create_audit_partition.sh >> logs/partition-create.log 2>&1

# Quarterly old partition cleanup (run on 1st of Jan/Apr/Jul/Oct at 3:00 AM)
# Removes partitions older than 90 days
0 3 1 1,4,7,10 * /path/to/modemcheck/cloudserver/scripts/cleanup_old_partitions.sh >> logs/partition-cleanup.log 2>&1


# ============================================================================
# AUDIT LOG CLEANUP
# ============================================================================

# Weekly audit log cleanup (every Sunday at 3:00 AM)
# Keeps 90 days of logs by default
0 3 * * 0 cd /path/to/modemcheck/cloudserver && python3 cleanup-audit-logs.py >> logs/cleanup.log 2>&1

# Alternative: Monthly cleanup with custom retention
# 0 3 1 * * cd /path/to/modemcheck/cloudserver && python3 cleanup-audit-logs.py --user-retention 60 --client-retention 120 >> logs/cleanup.log 2>&1
